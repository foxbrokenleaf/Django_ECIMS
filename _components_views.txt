103: def _category_subtree_ids(category_id):
104:     """返回品类自身及所有下级品类的 ID，用于列表按品类筛选。"""
105:     children_map = {}
106:     for pk, parent_id in Category.objects.values_list("id", "parent_id"):
107:         children_map.setdefault(parent_id, []).append(pk)
108: 
109:     ids, stack = [], [category_id]
110:     while stack:
111:         current = stack.pop()
112:         ids.append(current)
113:         stack.extend(children_map.get(current, []))
114:     return ids
